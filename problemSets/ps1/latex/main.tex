\section*{Problem 1}
\subsection*{a)}
We consider the state with basis $\ket{0}, \ket{1},\dots ,\ket{d-1}$, and have another orthonormal basis $\ket{v_0}, \ket{v_1}, \dots , \ket{v_{d-1}}$. To show that there is a unitary transformation $U$ such that $U\ket{i}=\ket{v_i}$ we start with an operator $\ket{v_i}\bra{i}$ and check how it acts on a vector $\ket{j}$.

\begin{equation}
    \left(\ket{v_i}\bra{i}\right) \ket{j} = \ket{v_i} \braket{i|j} = \braket{i|j}\ket{v_i} = \delta_{ij}\ket{v_i} 
\end{equation}
This gives that the operator $\ket{v_i}\bra{i}$ will return the vector $\ket{v_i}$ when applied to $\ket{i}$ and $0$ otherwise. Taking advantage of this attribute, we construct the transformation by taking 
\begin{equation}
    U = \sum_{i=0}^{d-1}\ket{v_i}\bra{i}
\end{equation}
To show that $U$ is a unitary operator we check 
\begin{equation*}
    UU^\dagger = \sum_{i=0}^{d-1}\ket{v_i}\bra{i}\sum_{j=0}^{d-1}\ket{j}\bra{v_j} = 
    \sum_{i,j=0}^{d-1}\ket{v_i}\braket{i|j}\bra{v_j} = 
    \sum_{i,j=0}^{d-1}\ket{v_i}\delta_{ij}\bra{v_j} = 
    \sum_{i=0}^{d-1}\ket{v_i}\bra{v_i} = I
\end{equation*}
\subsection*{b)}
We see what the operator does to an arbitrary vector of the basis: $\sum_{i=0}^{d-1}c_{i}\ket{v_i}$
\begin{equation}
    \sum_{i=0}^{d-1}\ket{v_i}\bra{v_i}\sum_{j=0}^{d-1}c_{j}\ket{v_j} = 
    \sum_{i=0}^{d-1}\sum_{j=0}^{d-1}c_{j}\ket{v_i}\braket{v_i|v_j} = 
    \sum_{i=0}^{d-1}\sum_{j=0}^{d-1}\delta_{ij}c_i\ket{v_i} = 
    \sum_{i=0}^{d-1}c_{i}\ket{v_i}\bra{v_i}
\end{equation}
i.e 
\begin{equation}
    \sum_{i=0}^{d-1}\ket{v_i}\bra{v_i} = I
\end{equation}

\section*{Problem 2}
\subsection*{a)}
Show that the matrix $U$ is unitary, i.e. $U^\dagger U = I$
\begin{equation}
    U = \frac{1}{\sqrt{2}}
    \left(
    \begin{matrix}
        1&i \\
        1&-i
    \end{matrix}
    \right)
\end{equation}

\begin{equation}
    U^\dagger = \frac{1}{\sqrt{2}}
    \left(
    \begin{matrix}
        1 & 1 \\
        -i & i
    \end{matrix}
    \right)
\end{equation}

\begin{equation}
    U^\dagger U = UU^\dagger = \frac{1}{{2}}
    \left(
        \begin{matrix}
            1*1 + i*\left(-i\right) & 1*1 + i*i \\ 
            1*1 + \left(-i\right)*\left(-i\right) & 1*1 + i*\left(-i\right)
        \end{matrix}
    \right)
     = I
\end{equation}

\subsection*{b)}
How U acts upon the states: 
$\ket{0}, \ket{1}, \ket{+}, \ket{-}, \ket{+i} and \ket{-i}$

    \end{equation}
    
    \begin{equation}

            \right)

\subsection*{c)}
A pure qubit has the density matrix $\rho = \frac{1}{2}\left(I+\boldsymbol{r}\cdot \boldsymbol{\sigma}\right)$, with $\boldsymbol{r} = \left(x,y,z\right)$ being its bloch vector.
From our results in the previous task, we see that $U$ rotates the Bloch sphere, $\left(x,y,z\right) \rightarrow \left(z,x,y\right)$. From this, we easily construcs the matrix:
\begin{equation}
    R = 
    \left(
    \begin{matrix}
        0&0&1 \\ 1&0&0 \\ 0&1&0
    \end{matrix}
    \right)
\end{equation}
We solve the equation $Rn = n$ to find the axis of rotation. 
\begin{equation}
    Rn = 
    \left(
    \begin{matrix}
        0&0&1 \\ 1&0&0 \\ 0&1&0
    \end{matrix}
    \right)
    \left(
        \begin{matrix}
            n_x \\ n_y \\ n_z
        \end{matrix}
    \right)
    =
    \left(
        \begin{matrix}
            n_x \\ n_y \\ n_z
        \end{matrix}
    \right)
\end{equation}
giving the solution $n = \frac{1}{\sqrt{3}}\left(1,1,1\right)$, i.e. axis $\left(1,1,1\right)$
\section*{Problem 3}
Let $\ket{0}, \ket{1},\ket{2}$ be a basis for the system.
\subsection*{a)}
Firstly we need to check the unit length of the basis.
Using the characteristic $\braket{x_i|x_j} = \delta_{ij},\forall\, x_{i,j} \in \ket{0},\ket{1},\ket{2}$ we get the following inner products:
\begin{equation}
    \braket{a|a} = \frac{1}{2}\braket{0|0} + \frac{1}{4}\braket{1|1} + \frac{1}{4}\braket{2|2} = \frac{1}{2} + \frac{1}{4} + \frac{1}{4} = 1
\end{equation}
\begin{equation}
    \braket{b|b} = \frac{1}{2}\braket{0|0} + \frac{1}{4}\braket{1|1} + \frac{1}{4}\braket{2|2} = \frac{1}{2} + \frac{1}{4} + \frac{1}{4} = 1
\end{equation}
\begin{equation}
    \braket{c|c} =  \frac{1}{2}\braket{1|1} + \frac{1}{2}\braket{2|2} = \frac{1}{2} + \frac{1}{2} = 1
\end{equation}
There can only exist three(3) orthonormal vectors in a three dimentional space, so we test inner products.
\begin{equation}
    \braket{a|b} = \frac{1}{2}\braket{0|0} - \frac{1}{4}\braket{1|1} - \frac{1}{4}\braket{2|2} = \frac{1}{2} - \frac{1}{4}-\frac{1}{4} = 0
\end{equation}

\begin{equation}
    \braket{b|c} = -\frac{\sqrt{2}-1}{2}\braket{1|1} + \frac{\sqrt{2}-1}{2}\braket{2|2} = -\frac{\sqrt{2}-1}{2}+\frac{\sqrt{2}-1}{2} = 0
\end{equation}

\begin{equation}
    \braket{a|c} = \frac{\sqrt{2}-1}{2}\braket{1|1} - \frac{\sqrt{2}-1}{2}\braket{2|2} = \frac{\sqrt{2}-1}{2}-\frac{\sqrt{2}-1}{2} = 0
\end{equation}
i.e. $\ket{a},\ket{b},\ket{c}$ is an orthonormal basis.

\subsection*{b)}
The von neumann method of measurement is using a self-adjoint operator named an "obeservable" to conduct the measurement. 
We take the projection of the state onto the basis:
\begin{equation}
    \braket{\psi|a} = \frac{\sqrt{2}}{\sqrt{6}}\braket{0|0} + \frac{\sqrt{2}}{\sqrt{6}}\braket{1|1} = \sqrt{\frac{2}{3}}
\end{equation}

\begin{equation}
    \braket{\psi|b} = \frac{\sqrt{2}}{\sqrt{6}}\braket{0|0} - \frac{\sqrt{2}}{\sqrt{6}}\braket{1|1} = 0
\end{equation}

\begin{equation}
    \braket{\psi|c} = \frac{\sqrt{2}}{\sqrt{6}}\braket{1|1} = \sqrt{\frac{1}{3}}
\end{equation}
I.e. we get probability $\frac{2}{3}$ of being in $\ket{a}$, $0$ for $\ket{b}$ and $\frac{1}{3}$ of being in state $\ket{c}$.

For measurement of $\ket{a},\ket{b} $ and $ \ket{c}$ the state collapses to subsequantly $\ket{a},\ket{b} $ and $ \ket{c}$

\section*{Problem 5}
\subsection*{a)}
We construct each object by combining a polarizing filter $P$ with a unitary transformation $U$. We want the unitary to have the property that light passing through the first polarizer and then through $U$ has probability $1/2$ of passing through any linear polarizer, independent of its orientation.

We choose our basis such that the first polarizer transmits

$$
    \ket{0}
    =
    \begin{pmatrix}
        1\\
        0
    \end{pmatrix}.
$$

Suppose that

$$
    U\ket{0}
    =
    \begin{pmatrix}
        a\\
        b
    \end{pmatrix}.
$$

A linear polarizer rotated by an arbitrary angle $\theta$ transmits the state

$$
    \ket{\theta}
    =
    \begin{pmatrix}
        \cos\theta\\
        \sin\theta
    \end{pmatrix}.
$$

We require the probability of passing through this polarizer to be $1/2$ for every $\theta$, such that

$$
    \left|\bra{\theta}U\ket{0}\right|^2
    =
    \left|a\cos\theta+b\sin\theta\right|^2
    =
    \frac{1}{2}.
$$

Expanding this expression gives

$$
    |a|^2\cos^2\theta
    +
    |b|^2\sin^2\theta
    +
    2\operatorname{Re}(a^*b)\sin\theta\cos\theta
    =
    \frac{1}{2}.
$$

For this to hold independently of $\theta$, we require

$$
    |a|^2=|b|^2=\frac{1}{2},
    \qquad
    \operatorname{Re}(a^*b)=0.
$$

A simple choice satisfying these conditions is

$$
    a=\frac{1}{\sqrt{2}},
    \qquad
    b=\frac{i}{\sqrt{2}},
$$

and hence

$$
    U\ket{0}
    =
    \frac{1}{\sqrt{2}}
    \begin{pmatrix}
        1\\
        i
    \end{pmatrix}.
$$

This determines the first column of $U$. Since $U$ must be unitary, its second column must be a normalized vector orthogonal to the first. One possible choice is

$$
    \frac{1}{\sqrt{2}}
    \begin{pmatrix}
        i\\
        1
    \end{pmatrix},
$$

giving

$$
    {
    U=
    \frac{1}{\sqrt{2}}
    \begin{pmatrix}
        1&i\\
        i&1
    \end{pmatrix}
    }.
$$

Indeed, $U^\dagger U=I$.

We now construct each object from one such unitary transformation and one polarizing filter. In the $ABCD$ configuration, the total transformation can be written as

$$
    U P_1 P_2 U^\dagger.
$$

The unitary transformations do not change the intensity, so the transmission probability is the same as for $P_1P_2$. Thus, the two objects behave like ordinary polarizing filters in this orientation.

If one object is flipped, the essential part of the transformation becomes

$$
    P_1 U P_2.
$$

After passing through the first polarizer, the photon is in the state $\ket{0}$ in the coordinate system of that object. Applying $U$ gives

$$
    U\ket{0}
    =
    \frac{1}{\sqrt{2}}
    \begin{pmatrix}
        1\\
        i
    \end{pmatrix}.
$$

The probability of this state passing through the second polarizer at an arbitrary relative angle $\theta$ is

$$
\begin{aligned}
    \left|\bra{\theta}U\ket{0}\right|^2
    &=
    \frac{1}{2}
    \left|\cos\theta+i\sin\theta\right|^2\\
    &=
    \frac{1}{2}
    \left(\cos^2\theta+\sin^2\theta\right)\\
    &=
    \frac{1}{2}.
\end{aligned}
$$

Thus, this probability is independent of the relative angle $\theta$. Since the first polarizer transmits $1/2$ of the initially unpolarized light, the total fraction transmitted is

$$
    \frac{1}{2}\cdot\frac{1}{2}
    =
    {\frac{1}{4}}.
$$

Therefore, this construction produces the required $25\%$ transmission independent of the relative orientation of the two objects.

\subsection*{b)}
If we order the objects as $BADC$, we would get the effect of $PUU^\dagger P$. Using the fact that $U$ is unitary, we just get $PP$, i.e. the effect of just two normal filters next to eachother
